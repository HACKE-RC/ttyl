(function () {
  "use strict";

  var id = location.pathname.split("/").filter(Boolean).pop();
  var terminalEl = document.getElementById("terminal");
  var metaEl = document.getElementById("meta");
  var commandEl = document.getElementById("command");
  var statusEl = document.getElementById("status");
  var playEl = document.getElementById("play");
  var restartEl = document.getElementById("restart");
  var speedEl = document.getElementById("speed");
  var copyEl = document.getElementById("copy");

  var archive = null;
  var events = [];
  var eventsDone = false;
  var term = null;
  var index = 0;
  var playing = true;
  var timer = 0;
  var DEFAULT_FONT_FAMILY =
    '"JetBrains Mono", "Symbols Nerd Font Mono", "JetBrainsMono Nerd Font", "JetBrainsMonoNL Nerd Font", "Symbols Nerd Font", ui-monospace, SFMono-Regular, Menlo, Consolas, monospace';
  var DEFAULT_FONT_SIZE = 15;

  function setStatus(text) {
    statusEl.textContent = text;
  }

  function decodeBase64(value) {
    var raw = atob(value);
    var bytes = new Uint8Array(raw.length);
    for (var i = 0; i < raw.length; i++) {
      bytes[i] = raw.charCodeAt(i);
    }
    return bytes;
  }

  function authHeaders() {
    var token = decodeURIComponent((location.hash || "").replace(/^#/, ""));
    if (token.indexOf("admin=") === 0) {
      token = token.slice("admin=".length);
    }
    return token ? { Authorization: "Bearer " + token } : {};
  }

  function bootTerminal() {
    if (term) {
      term.dispose();
    }
    var recording = archive.manifest.recording || {};
    var theme = xtermTheme(recording.theme || {});
    term = new Terminal({
      cols: archive.manifest.terminal.cols,
      rows: archive.manifest.terminal.rows,
      convertEol: false,
      cursorBlink: false,
      fontFamily: recording.fontFamily || DEFAULT_FONT_FAMILY,
      fontSize: recording.fontSize > 0 ? recording.fontSize : DEFAULT_FONT_SIZE,
      theme: theme,
    });
    terminalEl.textContent = "";
    term.open(terminalEl);
    if (document.fonts) {
      document.fonts.ready.then(function () {
        if (!term) {
          return;
        }
        term.options.fontFamily = recording.fontFamily || DEFAULT_FONT_FAMILY;
        term.options.fontSize = recording.fontSize > 0 ? recording.fontSize : DEFAULT_FONT_SIZE;
        term.refresh(0, term.rows - 1);
      });
    }
  }

  function xtermTheme(theme) {
    var ansi = Array.isArray(theme.ansi) ? theme.ansi : [];
    return {
      background: theme.background || "#0b0e14",
      foreground: theme.foreground || "#c8d3f5",
      cursor: theme.cursor || theme.foreground || "#c8d3f5",
      black: ansi[0],
      red: ansi[1],
      green: ansi[2],
      yellow: ansi[3],
      blue: ansi[4],
      magenta: ansi[5],
      cyan: ansi[6],
      white: ansi[7],
      brightBlack: ansi[8],
      brightRed: ansi[9],
      brightGreen: ansi[10],
      brightYellow: ansi[11],
      brightBlue: ansi[12],
      brightMagenta: ansi[13],
      brightCyan: ansi[14],
      brightWhite: ansi[15],
    };
  }

  function renderMeta() {
    var manifest = archive.manifest;
    var seconds = Math.max(0, Math.round(manifest.stats.durationMs / 1000));
    metaEl.textContent = manifest.id + " / " + manifest.status + " / " + seconds + "s";
    commandEl.textContent = manifest.command.join(" ");
  }

  function scheduleNext(previousElapsed) {
    clearTimeout(timer);
    timer = 0;
    if (!playing) {
      return;
    }
    if (index >= events.length) {
      if (!eventsDone) {
        setStatus("loading");
        return;
      }
      setStatus("done");
      playEl.textContent = "Play";
      playing = false;
      return;
    }
    var speed = Number(speedEl.value) || 1;
    var event = events[index];
    var delay = Math.max(0, event.elapsedMs - previousElapsed) / speed;
    timer = window.setTimeout(function () {
      timer = 0;
      if (event.type === "data") {
        term.write(decodeBase64(event.data));
      } else if (event.type === "resize") {
        term.resize(event.cols, event.rows);
      }
      index++;
      setStatus(index + "/" + events.length);
      scheduleNext(event.elapsedMs);
    }, delay);
  }

  function restart() {
    index = 0;
    playing = true;
    playEl.textContent = "Pause";
    bootTerminal();
    scheduleNext(0);
  }

  playEl.addEventListener("click", function () {
    playing = !playing;
    playEl.textContent = playing ? "Pause" : "Play";
    if (playing) {
      var previous = index > 0 && events[index - 1] ? events[index - 1].elapsedMs : 0;
      scheduleNext(previous);
    } else {
      clearTimeout(timer);
    }
  });

  restartEl.addEventListener("click", restart);
  speedEl.addEventListener("change", function () {
    if (playing) {
      var previous = index > 0 && events[index - 1] ? events[index - 1].elapsedMs : 0;
      scheduleNext(previous);
    }
  });
  copyEl.addEventListener("click", function () {
    fetch("/api/vaults/" + encodeURIComponent(id) + "/transcript", {
      cache: "no-store",
      headers: authHeaders(),
    })
      .then(function (response) {
        if (!response.ok) throw new Error("transcript not found");
        return response.text();
      })
      .then(function (transcript) {
        return navigator.clipboard.writeText(transcript);
      })
      .then(function () {
        setStatus("copied");
      })
      .catch(function (err) {
        setStatus(err.message || String(err));
      });
  });

  fetch("/api/vaults/" + encodeURIComponent(id), { cache: "no-store", headers: authHeaders() })
    .then(function (response) {
      if (!response.ok) {
        throw new Error("vault not found");
      }
      return response.json();
    })
    .then(function (payload) {
      archive = payload;
      bootTerminal();
      renderMeta();
      setStatus("loading");
      streamEvents();
    })
    .catch(function (err) {
      setStatus(err.message || String(err));
    });

  function streamEvents() {
    fetch("/api/vaults/" + encodeURIComponent(id) + "/events", {
      cache: "no-store",
      headers: authHeaders(),
    })
      .then(function (response) {
        if (!response.ok) throw new Error("events not found");
        if (!response.body || !response.body.getReader) {
          return response.text().then(function (text) {
            appendEventLines(text);
            eventsDone = true;
            scheduleNext(index > 0 && events[index - 1] ? events[index - 1].elapsedMs : 0);
          });
        }
        var reader = response.body.getReader();
        var decoder = new TextDecoder();
        var pending = "";
        function pump() {
          return reader.read().then(function (result) {
            if (result.done) {
              appendEventLines(pending);
              eventsDone = true;
              scheduleNext(index > 0 && events[index - 1] ? events[index - 1].elapsedMs : 0);
              return;
            }
            pending += decoder.decode(result.value, { stream: true });
            var cut = pending.lastIndexOf("\n");
            if (cut !== -1) {
              appendEventLines(pending.slice(0, cut + 1));
              pending = pending.slice(cut + 1);
              if (playing && !timer) {
                scheduleNext(index > 0 && events[index - 1] ? events[index - 1].elapsedMs : 0);
              }
            }
            return pump();
          });
        }
        return pump();
      })
      .catch(function (err) {
        setStatus(err.message || String(err));
      });
  }

  function appendEventLines(text) {
    text
      .split(/\n/)
      .filter(Boolean)
      .forEach(function (line) {
        events.push(JSON.parse(line));
      });
  }
})();
