// ttyl web viewer: renders the source PTY grid and scales it visually.
//
// Terminal control sequences are interpreted only at the source PTY size. The
// browser must not replay them into a different grid because TUIs use absolute
// cursor positions and wrap decisions based on the source terminal width.
const KIND_OUTPUT = 0;
const KIND_INPUT = 1;
const KIND_RESIZE = 2;
const KIND_AUTH = 3;
const CONTROL_KEY_STORAGE_PREFIX = "ttyl.viewer.controlKey.";

const statusEl = document.getElementById("status");
const statusText = document.getElementById("status-text");
const fitEl = document.getElementById("fit");
const viewportEl = document.getElementById("viewport");
const stageEl = document.getElementById("stage");
const termEl = document.getElementById("terminal");
const zoomOutEl = document.getElementById("zoom-out");
const zoomInEl = document.getElementById("zoom-in");
const modeFitEl = document.getElementById("mode-fit");
const modeReadableEl = document.getElementById("mode-readable");
const modeActualEl = document.getElementById("mode-actual");
const scaleLabelEl = document.getElementById("scale-label");
const TERMINAL_FONT =
  '"JetBrains Mono", "Symbols Nerd Font Mono", "JetBrainsMono Nerd Font", "JetBrainsMonoNL Nerd Font", "Symbols Nerd Font", ui-monospace, SFMono-Regular, Menlo, Consolas, monospace';
const BASE_FONT_SIZE = 15;
const MODE_FIT = "fit";
const MODE_READABLE = "readable";
const MODE_ACTUAL = "actual";
const MIN_READABLE_CELL_WIDTH = 7;
const MIN_READABLE_CELL_HEIGHT = 12;
const ZOOM_STEP = 1.15;
const MIN_USER_ZOOM = 0.25;
const MAX_USER_ZOOM = 5;

let canType = false;
let socket;
let stopped = false;
let layoutFrame = 0;
let userZoom = 1;
let scaleMode = MODE_READABLE;
let xOffset = 0;
let yOffset = 0;
let lastLayout = null;
let pointerDrag = null;

function setStatus(text, live) {
  statusText.textContent = text;
  statusEl.classList.toggle("live", Boolean(live));
}

const term = new Terminal({
  cursorBlink: true,
  fontFamily: TERMINAL_FONT,
  fontSize: BASE_FONT_SIZE,
  theme: { background: "#0b0e14", foreground: "#c8d3f5" },
});
term.open(termEl);
term.focus();
if (document.fonts) {
  document.fonts.ready.then(() => {
    term.options.fontFamily = TERMINAL_FONT;
    term.refresh(0, term.rows - 1);
    scheduleLayout();
  });
}
term.onResize(scheduleLayout);
term.onRender(scheduleLayout);
window.addEventListener("resize", scheduleLayout);
if (window.visualViewport) {
  window.visualViewport.addEventListener("resize", scheduleLayout);
}
if (window.ResizeObserver) {
  new ResizeObserver(scheduleLayout).observe(fitEl);
}
viewportEl.addEventListener("wheel", handleWheel, { passive: false });
viewportEl.addEventListener("pointerdown", handlePointerDown);
viewportEl.addEventListener("pointermove", handlePointerMove);
viewportEl.addEventListener("pointerup", handlePointerUp);
viewportEl.addEventListener("pointercancel", handlePointerUp);
zoomOutEl.addEventListener("click", () => setZoom(userZoom / ZOOM_STEP));
zoomInEl.addEventListener("click", () => setZoom(userZoom * ZOOM_STEP));
modeFitEl.addEventListener("click", () => setScaleMode(MODE_FIT));
modeReadableEl.addEventListener("click", () => setScaleMode(MODE_READABLE));
modeActualEl.addEventListener("click", () => setScaleMode(MODE_ACTUAL));

const pathParts = location.pathname.split("/").filter(Boolean);
const sessionId = pathParts[1];
const controlKey = loadControlKey();
canType = controlKey.length > 0;

const encoder = new TextEncoder();
function sendInput(data) {
  const bytes = encoder.encode(data);
  const frame = new Uint8Array(1 + bytes.length);
  frame[0] = KIND_INPUT;
  frame.set(bytes, 1);
  if (socket && socket.readyState === WebSocket.OPEN) {
    socket.send(frame);
  }
}
if (canType) {
  term.onData(sendInput);
} else {
  term.options.disableStdin = true;
}

function handleFrame(buffer) {
  const view = new DataView(buffer);
  const kind = view.getUint8(0);
  if (kind === KIND_OUTPUT) {
    term.write(new Uint8Array(buffer, 1));
  } else if (kind === KIND_RESIZE) {
    const cols = view.getUint16(1, false);
    const rows = view.getUint16(3, false);
    if (cols > 0 && rows > 0) {
      term.resize(cols, rows);
      scheduleLayout();
    }
  }
}

function scheduleLayout() {
  if (layoutFrame) return;
  layoutFrame = requestAnimationFrame(() => {
    layoutFrame = 0;
    applyLayout();
  });
}

function applyLayout() {
  const natural = terminalNaturalSize();
  if (!natural.width || !natural.height) return;
  const box = viewerBox();
  if (!box.width || !box.height) return;
  const scales = scaleChoices(natural, box);
  const baseScale = scales[scaleMode] ?? scales[MODE_READABLE];
  const scale = Math.max(scales[MODE_FIT], baseScale * userZoom);
  const maxX = Math.max(0, natural.width * scale - box.width);
  const maxY = Math.max(0, natural.height * scale - box.height);
  xOffset = clamp(xOffset, 0, maxX);
  yOffset = clamp(yOffset, 0, maxY);
  viewportEl.style.width = `${box.width}px`;
  viewportEl.style.height = `${box.height}px`;
  stageEl.style.width = `${natural.width}px`;
  stageEl.style.height = `${natural.height}px`;
  stageEl.style.left = `${-xOffset}px`;
  stageEl.style.top = `${-yOffset}px`;
  stageEl.style.transform = `scale(${scale})`;
  lastLayout = { natural, box, baseScale, scale, maxX, maxY, coverScale: scales[MODE_FIT] };
  updateControls(scale);
}

function scaleChoices(natural, box) {
  const widthFit = box.width / natural.width;
  const heightFit = box.height / natural.height;
  const containScale = Math.min(widthFit, heightFit);
  const cols = term.cols || 1;
  const rows = term.rows || 1;
  const cellWidth = natural.width / cols;
  const cellHeight = natural.height / rows;
  const readableTarget = Math.max(
    MIN_READABLE_CELL_WIDTH / cellWidth,
    MIN_READABLE_CELL_HEIGHT / cellHeight,
  );
  // Default to a legible view only when it can still show the whole vertical
  // grid. Full-screen TUIs keep prompts/status bars at the bottom; hiding them
  // by default is worse than needing horizontal pan.
  const readableScale = clamp(readableTarget, containScale, heightFit);
  return {
    [MODE_FIT]: containScale,
    [MODE_READABLE]: readableScale,
    [MODE_ACTUAL]: Math.max(containScale, 1),
  };
}

function handleWheel(event) {
  if (!lastLayout) scheduleLayout();
  const layout = lastLayout;
  if (!layout) return;

  if (event.ctrlKey || event.metaKey) {
    event.preventDefault();
    const rect = viewportEl.getBoundingClientRect();
    const px = event.clientX - rect.left;
    const py = event.clientY - rect.top;
    const beforeX = (xOffset + px) / layout.scale;
    const beforeY = (yOffset + py) / layout.scale;
    const factor = Math.exp(-event.deltaY * 0.0015);
    userZoom = clamp(userZoom * factor, MIN_USER_ZOOM, MAX_USER_ZOOM);
    const nextScale = layout.baseScale * userZoom;
    xOffset = beforeX * nextScale - px;
    yOffset = beforeY * nextScale - py;
    applyLayout();
    return;
  }

  if (layout.maxX > 0 || layout.maxY > 0) {
    event.preventDefault();
    xOffset += event.shiftKey && Math.abs(event.deltaY) > Math.abs(event.deltaX) ? event.deltaY : event.deltaX;
    yOffset += event.deltaY;
    applyLayout();
  }
}

function setZoom(nextZoom) {
  userZoom = clamp(nextZoom, MIN_USER_ZOOM, MAX_USER_ZOOM);
  applyLayout();
}

function setScaleMode(nextMode) {
  scaleMode = nextMode;
  userZoom = 1;
  xOffset = 0;
  yOffset = 0;
  applyLayout();
}

function updateControls(scale) {
  scaleLabelEl.textContent = `${Math.round(scale * 100)}%`;
  modeFitEl.classList.toggle("active", scaleMode === MODE_FIT);
  modeReadableEl.classList.toggle("active", scaleMode === MODE_READABLE);
  modeActualEl.classList.toggle("active", scaleMode === MODE_ACTUAL);
}

function handlePointerDown(event) {
  if (!lastLayout || (lastLayout.maxX === 0 && lastLayout.maxY === 0)) return;
  pointerDrag = { id: event.pointerId, x: event.clientX, y: event.clientY };
  viewportEl.setPointerCapture(event.pointerId);
}

function handlePointerMove(event) {
  if (!pointerDrag || pointerDrag.id !== event.pointerId) return;
  event.preventDefault();
  xOffset -= event.clientX - pointerDrag.x;
  yOffset -= event.clientY - pointerDrag.y;
  pointerDrag.x = event.clientX;
  pointerDrag.y = event.clientY;
  applyLayout();
}

function handlePointerUp(event) {
  if (!pointerDrag || pointerDrag.id !== event.pointerId) return;
  pointerDrag = null;
  try {
    viewportEl.releasePointerCapture(event.pointerId);
  } catch {
    // ignore stale pointer captures
  }
}

function terminalNaturalSize() {
  const screen = termEl.querySelector(".xterm-screen");
  const el = screen || termEl;
  const rect = el.getBoundingClientRect();
  return {
    width: el.scrollWidth || rect.width,
    height: el.scrollHeight || rect.height,
  };
}

function viewerBox() {
  const style = getComputedStyle(fitEl);
  const paddingX = parseFloat(style.paddingLeft) + parseFloat(style.paddingRight);
  const paddingY = parseFloat(style.paddingTop) + parseFloat(style.paddingBottom);
  return {
    width: Math.max(0, fitEl.clientWidth - paddingX),
    height: Math.max(0, fitEl.clientHeight - paddingY),
  };
}

function clamp(value, min, max) {
  return Math.max(min, Math.min(max, value));
}

const wsScheme = location.protocol === "https:" ? "wss" : "ws";
const wsURL = `${wsScheme}://${location.host}/ws/${sessionId}/view`;
const liveLabel = canType ? "live" : "live (view-only)";

const CLOSE_KICKED = 4001;
const CLOSE_LOCKED = 4002;
const CLOSE_PASSWORD_REQUIRED = 4003;
const CLOSE_PASSWORD_INCORRECT = 4004;
const CLOSE_TOO_MANY_ATTEMPTS = 4005;

let password = "";

function sendAuth() {
  const payload = {};
  if (controlKey) payload.k = controlKey;
  if (password) payload.p = password;
  const bytes = encoder.encode(JSON.stringify(payload));
  const frame = new Uint8Array(1 + bytes.length);
  frame[0] = KIND_AUTH;
  frame.set(bytes, 1);
  socket.send(frame);
}

function askPassword(message) {
  return new Promise((resolve) => {
    const overlay = document.createElement("div");
    overlay.setAttribute(
      "style",
      "position:fixed;inset:0;display:flex;align-items:center;justify-content:center;background:rgba(11,14,20,0.92);z-index:10;",
    );
    const box = document.createElement("form");
    box.setAttribute(
      "style",
      "display:flex;flex-direction:column;gap:10px;min-width:260px;padding:20px;border:1px solid #1b2230;border-radius:8px;background:#11141c;",
    );
    const label = document.createElement("label");
    label.textContent = message;
    label.setAttribute("style", "font-size:13px;color:#c8d3f5;");
    const input = document.createElement("input");
    input.type = "password";
    input.autofocus = true;
    input.setAttribute(
      "style",
      "padding:8px;border:1px solid #1b2230;border-radius:4px;background:#0b0e14;color:#c8d3f5;font-family:inherit;",
    );
    const button = document.createElement("button");
    button.type = "submit";
    button.textContent = "Unlock";
    button.setAttribute(
      "style",
      "padding:8px;border:none;border-radius:4px;background:#9ece6a;color:#0b0e14;font-weight:600;cursor:pointer;",
    );
    box.appendChild(label);
    box.appendChild(input);
    box.appendChild(button);
    overlay.appendChild(box);
    box.addEventListener("submit", (e) => {
      e.preventDefault();
      const value = input.value;
      overlay.remove();
      resolve(value);
    });
    document.body.appendChild(overlay);
    input.focus();
  });
}

function loadControlKey() {
  const fragmentKey = location.hash ? decodeURIComponent(location.hash.replace(/^#/, "")) : "";
  const storageKey = `${CONTROL_KEY_STORAGE_PREFIX}${sessionId}`;
  if (fragmentKey) {
    try {
      window.sessionStorage.setItem(storageKey, fragmentKey);
    } catch {
      // Keep the key in memory if tab-scoped storage is unavailable.
    }
    history.replaceState(null, "", location.pathname + location.search);
    return fragmentKey;
  }
  if (!isReloadNavigation()) {
    return "";
  }
  try {
    return window.sessionStorage.getItem(storageKey) ?? "";
  } catch {
    return "";
  }
}

function isReloadNavigation() {
  const nav = performance.getEntriesByType?.("navigation")?.[0];
  return nav && "type" in nav ? nav.type === "reload" : performance.navigation?.type === 1;
}

function connect() {
  if (stopped) return;
  setStatus("connecting...", false);
  socket = new WebSocket(wsURL);
  socket.binaryType = "arraybuffer";

  socket.onopen = () => {
    sendAuth();
    setStatus(liveLabel, true);
  };
  socket.onmessage = (event) => handleFrame(event.data);
  socket.onclose = (event) => handleClose(event.code);
  socket.onerror = () => socket.close();
}

function handleClose(code) {
  switch (code) {
    case CLOSE_KICKED:
      stopped = true;
      setStatus("removed by host", false);
      return;
    case CLOSE_LOCKED:
      stopped = true;
      setStatus("session locked - no new viewers", false);
      return;
    case CLOSE_PASSWORD_REQUIRED:
      promptAndReconnect("This session is password protected:");
      return;
    case CLOSE_PASSWORD_INCORRECT:
      promptAndReconnect("Incorrect password, try again:");
      return;
    case CLOSE_TOO_MANY_ATTEMPTS:
      setStatus("too many attempts - retrying shortly...", false);
      setTimeout(connect, 15000);
      return;
    case 1000:
      stopped = true;
      setStatus("session ended", false);
      return;
    default:
      setStatus("disconnected - retrying...", false);
      setTimeout(connect, 1500);
  }
}

function promptAndReconnect(message) {
  setStatus("password required", false);
  askPassword(message).then((value) => {
    password = value;
    connect();
  });
}

connect();
scheduleLayout();
term.focus();
