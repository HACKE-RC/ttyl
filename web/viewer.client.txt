// ttyl web viewer: renders a live PTY stream and forwards keystrokes.
//
// The wire protocol mirrors internal/wire: each WebSocket binary message is a
// single frame whose first byte is the kind, followed by the payload.
//   0 = Output  (PTY bytes to render)
//   1 = Input   (keystroke bytes to send)
//   2 = Resize  (uint16 cols, uint16 rows, big-endian)
const KIND_OUTPUT = 0;
const KIND_INPUT = 1;
const KIND_RESIZE = 2;
const KIND_AUTH = 3;
const CONTROL_KEY_STORAGE_PREFIX = "ttyl.viewer.controlKey.";

const statusEl = document.getElementById("status");
const statusText = document.getElementById("status-text");
const zoomOutEl = document.getElementById("zoom-out");
const zoomInEl = document.getElementById("zoom-in");
const zoomResetEl = document.getElementById("zoom-reset");
const scaleLabelEl = document.getElementById("scale-label");
const fitEl = document.getElementById("fit");
const viewportEl = document.getElementById("viewport");
const termEl = document.getElementById("terminal");
const TERMINAL_FONT =
  '"JetBrains Mono", "Symbols Nerd Font Mono", "JetBrainsMono Nerd Font", "JetBrainsMonoNL Nerd Font", "Symbols Nerd Font", ui-monospace, SFMono-Regular, Menlo, Consolas, monospace';
const BASE_FONT_SIZE = 15;
const DEFAULT_SCALE = 1;
const MIN_SCALE = 0.1;
const MAX_SCALE = 4;
const SCALE_STEP = 0.1;
const MIN_VIEWER_COLS = 20;
const MIN_VIEWER_ROWS = 5;
const MAX_VIEWER_COLS = 500;
const MAX_VIEWER_ROWS = 200;
const MODE_MANUAL = "manual";
const MODE_FIT_WIDTH = "fit-width";
let viewerScale = DEFAULT_SCALE;
let scaleMode = MODE_FIT_WIDTH;
let currentScale = viewerScale;
let sourceCols = 0;
let sourceRows = 0;
let layoutFrame = 0;
let browserResizeFrame = 0;
let lastSentResizeKey = "";
let canType = false;
let socket;
let stopped = false;

function setStatus(text, live) {
  statusText.textContent = text;
  statusEl.classList.toggle("live", Boolean(live));
}

const term = new Terminal({
  cursorBlink: true,
  fontFamily: TERMINAL_FONT,
  fontSize: BASE_FONT_SIZE,
  theme: { background: "#0b0e14", foreground: "#c8d3f5" },
});
term.open(termEl);
term.focus();
if (document.fonts) {
  document.fonts.ready.then(() => {
    term.options.fontFamily = TERMINAL_FONT;
    term.refresh(0, term.rows - 1);
    scheduleViewportChange();
  });
}
wireZoomControls();
term.onResize(scheduleViewportChange);
term.onRender(scheduleLayout);
window.addEventListener("resize", scheduleViewportChange);
if (window.visualViewport) {
  window.visualViewport.addEventListener("resize", scheduleViewportChange);
}
if (window.ResizeObserver) {
  new ResizeObserver(scheduleViewportChange).observe(fitEl);
}

// The page is always /s/{id}. The read-write control key, if any, arrives in
// the URL fragment (#{key}), which the browser never sends to the server. A
// present key authorizes typing; without it this viewer is view-only. The key
// is read once and then stripped from the address bar so it does not linger in
// the visible URL or this navigation's history entry.
const pathParts = location.pathname.split("/").filter(Boolean);
const sessionId = pathParts[1];
const controlKey = loadControlKey();
canType = controlKey.length > 0;

// The source PTY has one canonical grid. A read-write browser is allowed to
// drive that grid from its viewport, like a normal web terminal; view-only
// browsers only mirror the grid the broadcaster confirms.
const encoder = new TextEncoder();
function sendInput(data) {
  const bytes = encoder.encode(data);
  const frame = new Uint8Array(1 + bytes.length);
  frame[0] = KIND_INPUT;
  frame.set(bytes, 1);
  if (socket && socket.readyState === WebSocket.OPEN) {
    socket.send(frame);
  }
}
if (canType) {
  term.onData(sendInput);
} else {
  term.options.disableStdin = true;
}

function handleFrame(buffer) {
  const view = new DataView(buffer);
  const kind = view.getUint8(0);
  if (kind === KIND_OUTPUT) {
    term.write(new Uint8Array(buffer, 1));
  } else if (kind === KIND_RESIZE) {
    const cols = view.getUint16(1, false);
    const rows = view.getUint16(3, false);
    if (cols > 0 && rows > 0) {
      sourceCols = cols;
      sourceRows = rows;
      term.resize(cols, rows);
      scheduleViewportChange();
    }
  }
}

const wsScheme = location.protocol === "https:" ? "wss" : "ws";
const wsURL = `${wsScheme}://${location.host}/ws/${sessionId}/view`;
const liveLabel = canType ? "live" : "live (view-only)";

// Application close codes the relay uses to explain a rejected/ended viewer.
// Keep these in sync with the CLOSE_* constants in src/core/relay.ts.
const CLOSE_KICKED = 4001;
const CLOSE_LOCKED = 4002;
const CLOSE_PASSWORD_REQUIRED = 4003;
const CLOSE_PASSWORD_INCORRECT = 4004;
const CLOSE_TOO_MANY_ATTEMPTS = 4005;

// The session password, supplied by the viewer when the relay asks for it. It
// is held only in memory for this page and sent in the Auth frame, never put in
// the URL.
let password = "";

// The control key and password never go in the URL; they are proven with an
// Auth frame (JSON { k?, p? }) sent as the first message after the socket opens.
// An empty object means a view-only viewer with no password.
function sendAuth() {
  const payload = {};
  if (controlKey) payload.k = controlKey;
  if (password) payload.p = password;
  const bytes = encoder.encode(JSON.stringify(payload));
  const frame = new Uint8Array(1 + bytes.length);
  frame[0] = KIND_AUTH;
  frame.set(bytes, 1);
  socket.send(frame);
}

// A minimal password prompt overlay, built in JS so the page ships no extra
// markup when no password is set. The promise resolves with the entered value.
function askPassword(message) {
  return new Promise((resolve) => {
    const overlay = document.createElement("div");
    overlay.setAttribute(
      "style",
      "position:fixed;inset:0;display:flex;align-items:center;justify-content:center;background:rgba(11,14,20,0.92);z-index:10;",
    );
    const box = document.createElement("form");
    box.setAttribute(
      "style",
      "display:flex;flex-direction:column;gap:10px;min-width:260px;padding:20px;border:1px solid #1b2230;border-radius:8px;background:#11141c;",
    );
    const label = document.createElement("label");
    label.textContent = message;
    label.setAttribute("style", "font-size:13px;color:#c8d3f5;");
    const input = document.createElement("input");
    input.type = "password";
    input.autofocus = true;
    input.setAttribute(
      "style",
      "padding:8px;border:1px solid #1b2230;border-radius:4px;background:#0b0e14;color:#c8d3f5;font-family:inherit;",
    );
    const button = document.createElement("button");
    button.type = "submit";
    button.textContent = "Unlock";
    button.setAttribute(
      "style",
      "padding:8px;border:none;border-radius:4px;background:#9ece6a;color:#0b0e14;font-weight:600;cursor:pointer;",
    );
    box.appendChild(label);
    box.appendChild(input);
    box.appendChild(button);
    overlay.appendChild(box);
    box.addEventListener("submit", (e) => {
      e.preventDefault();
      const value = input.value;
      overlay.remove();
      resolve(value);
    });
    document.body.appendChild(overlay);
    input.focus();
  });
}

function loadControlKey() {
  const fragmentKey = location.hash ? decodeURIComponent(location.hash.replace(/^#/, "")) : "";
  const storageKey = `${CONTROL_KEY_STORAGE_PREFIX}${sessionId}`;
  if (fragmentKey) {
    try {
      window.sessionStorage.setItem(storageKey, fragmentKey);
    } catch {
      // Keep the key in memory if tab-scoped storage is unavailable.
    }
    history.replaceState(null, "", location.pathname + location.search);
    return fragmentKey;
  }
  if (!isReloadNavigation()) {
    return "";
  }
  try {
    return window.sessionStorage.getItem(storageKey) ?? "";
  } catch {
    return "";
  }
}

function isReloadNavigation() {
  const nav = performance.getEntriesByType?.("navigation")?.[0];
  return nav && "type" in nav ? nav.type === "reload" : performance.navigation?.type === 1;
}

function clampScale(value) {
  const rounded = Math.round(value * 100) / 100;
  return Math.max(MIN_SCALE, Math.min(MAX_SCALE, rounded));
}

function setViewerScale(next) {
  const clamped = clampScale(next);
  if (clamped === viewerScale && scaleMode === MODE_MANUAL) return;
  viewerScale = clamped;
  scaleMode = MODE_MANUAL;
  scheduleLayout();
}

function setScaleMode(next) {
  if (scaleMode === next) return;
  scaleMode = next;
  scheduleLayout();
}

function scheduleViewportChange() {
  scheduleLayout();
  scheduleBrowserResize();
}

function scheduleLayout() {
  if (layoutFrame) return;
  layoutFrame = requestAnimationFrame(() => {
    layoutFrame = 0;
    applyLayout();
  });
}

function applyLayout() {
  const { width: naturalWidth, height: naturalHeight } = terminalNaturalSize();
  if (!naturalWidth || !naturalHeight) return;
  const box = viewerBox();
  const nextScale =
    scaleMode === MODE_FIT_WIDTH
      ? clampScale(Math.min(box.width / naturalWidth, box.height / naturalHeight))
      : viewerScale;
  termEl.style.transform = `scale(${nextScale})`;
  viewportEl.style.width = `${Math.max(box.width, Math.ceil(naturalWidth * nextScale))}px`;
  viewportEl.style.height = `${Math.max(box.height, Math.ceil(naturalHeight * nextScale))}px`;
  currentScale = nextScale;
  updateScaleLabel();
}

function scheduleBrowserResize() {
  if (!canType || browserResizeFrame) return;
  browserResizeFrame = requestAnimationFrame(() => {
    browserResizeFrame = 0;
    sendBrowserResize();
  });
}

function sendBrowserResize() {
  if (!canType || !socket || socket.readyState !== WebSocket.OPEN) return;
  const grid = desiredBrowserGrid();
  if (!grid) return;
  const key = `${grid.cols}x${grid.rows}`;
  if (key === lastSentResizeKey || (grid.cols === sourceCols && grid.rows === sourceRows)) {
    return;
  }
  const frame = new Uint8Array(5);
  const view = new DataView(frame.buffer);
  frame[0] = KIND_RESIZE;
  view.setUint16(1, grid.cols, false);
  view.setUint16(3, grid.rows, false);
  socket.send(frame);
  lastSentResizeKey = key;
}

function desiredBrowserGrid() {
  const { width: naturalWidth, height: naturalHeight } = terminalNaturalSize();
  const cols = term.cols || sourceCols;
  const rows = term.rows || sourceRows;
  if (!naturalWidth || !naturalHeight || !cols || !rows) return null;
  const cellWidth = naturalWidth / cols;
  const cellHeight = naturalHeight / rows;
  if (!Number.isFinite(cellWidth) || !Number.isFinite(cellHeight) || cellWidth <= 0 || cellHeight <= 0) {
    return null;
  }
  const box = viewerBox();
  return {
    cols: clampInt(Math.floor(box.width / cellWidth), MIN_VIEWER_COLS, MAX_VIEWER_COLS),
    rows: clampInt(Math.floor(box.height / cellHeight), MIN_VIEWER_ROWS, MAX_VIEWER_ROWS),
  };
}

function terminalNaturalSize() {
  const screen = termEl.querySelector(".xterm-screen");
  const el = screen || termEl;
  const rect = el.getBoundingClientRect();
  return {
    width: el.scrollWidth || rect.width,
    height: el.scrollHeight || rect.height,
  };
}

function clampInt(value, min, max) {
  return Math.max(min, Math.min(max, value));
}

function wireZoomControls() {
  scheduleLayout();
  zoomOutEl.addEventListener("click", () => setViewerScale(currentScale - SCALE_STEP));
  zoomInEl.addEventListener("click", () => setViewerScale(currentScale + SCALE_STEP));
  zoomResetEl.addEventListener("click", () => setScaleMode(MODE_FIT_WIDTH));
  scaleLabelEl.addEventListener("click", () => setViewerScale(DEFAULT_SCALE));
  termEl.addEventListener("click", () => term.focus());
  fitEl.addEventListener(
    "wheel",
    (event) => {
      if (!event.ctrlKey) return;
      event.preventDefault();
      setViewerScale(currentScale + (event.deltaY < 0 ? SCALE_STEP : -SCALE_STEP));
    },
    { passive: false },
  );
}

function viewerBox() {
  const style = getComputedStyle(fitEl);
  const paddingX = parseFloat(style.paddingLeft) + parseFloat(style.paddingRight);
  const paddingY = parseFloat(style.paddingTop) + parseFloat(style.paddingBottom);
  return {
    width: Math.max(0, fitEl.clientWidth - paddingX),
    height: Math.max(0, fitEl.clientHeight - paddingY),
  };
}

function updateScaleLabel() {
  const percent = Math.round(currentScale * 100);
  const grid = sourceCols && sourceRows ? ` ${sourceCols}x${sourceRows}` : "";
  scaleLabelEl.textContent = scaleMode === MODE_FIT_WIDTH ? `fit ${percent}%${grid}` : `${percent}%${grid}`;
  zoomResetEl.classList.toggle("active", scaleMode === MODE_FIT_WIDTH);
}

function connect() {
  if (stopped) return;
  setStatus("connecting...", false);
  socket = new WebSocket(wsURL);
  socket.binaryType = "arraybuffer";

  socket.onopen = () => {
    lastSentResizeKey = "";
    sendAuth();
    setStatus(liveLabel, true);
    scheduleBrowserResize();
  };
  socket.onmessage = (event) => handleFrame(event.data);
  socket.onclose = (event) => handleClose(event.code);
  socket.onerror = () => socket.close();
}

function handleClose(code) {
  switch (code) {
    case CLOSE_KICKED:
      stopped = true;
      setStatus("removed by host", false);
      return;
    case CLOSE_LOCKED:
      stopped = true;
      setStatus("session locked - no new viewers", false);
      return;
    case CLOSE_PASSWORD_REQUIRED:
      promptAndReconnect("This session is password protected:");
      return;
    case CLOSE_PASSWORD_INCORRECT:
      promptAndReconnect("Incorrect password, try again:");
      return;
    case CLOSE_TOO_MANY_ATTEMPTS:
      setStatus("too many attempts - retrying shortly...", false);
      setTimeout(connect, 15000);
      return;
    case 1000:
      stopped = true;
      setStatus("session ended", false);
      return;
    default:
      setStatus("disconnected - retrying...", false);
      setTimeout(connect, 1500);
  }
}

function promptAndReconnect(message) {
  setStatus("password required", false);
  askPassword(message).then((value) => {
    password = value;
    connect();
  });
}

connect();
term.focus();
